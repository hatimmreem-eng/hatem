    <script>
        // Global variables
        var records = [];
        var currentFiles = [];
        var stream = null;
        var currentPhotoIndex = 0;
        var capturedPhotos = [null, null, null];

        // Initialize
        window.onload = function() {
            try {
                var saved = localStorage.getItem('criminalRecords');
                if (saved) {
                    records = JSON.parse(saved);
                }
            } catch(e) {
                records = [];
            }

            // Setup enter key
            var passwordInput = document.getElementById('password');
            var usernameInput = document.getElementById('username');

            if (passwordInput) {
                passwordInput.addEventListener('keydown', function(e) {
                    if (e.key === 'Enter') {
                        e.preventDefault();
                        doLogin();
                    }
                });
            }

            if (usernameInput) {
                usernameInput.addEventListener('keydown', function(e) {
                    if (e.key === 'Enter') {
                        e.preventDefault();
                        if (passwordInput) passwordInput.focus();
                    }
                });
            }
        };

        // Login function
        function doLogin() {
            var usernameInput = document.getElementById('username');
            var passwordInput = document.getElementById('password');
            var username = usernameInput ? usernameInput.value.trim() : '';
            var password = passwordInput ? passwordInput.value.trim() : '';
            var errorMsg = document.getElementById('errorMsg');

            if (username === 'ali@11' && password === '198922') {
                var loginScreen = document.getElementById('loginScreen');
                var mainApp = document.getElementById('mainApp');

                if (loginScreen) loginScreen.classList.add('hidden');
                if (mainApp) mainApp.classList.remove('hidden');
                if (errorMsg) errorMsg.style.display = 'none';

                loadRecords();
            } else {
                if (errorMsg) {
                    errorMsg.style.display = 'block';
                    errorMsg.textContent = 'اسم المستخدم أو كلمة المرور غير صحيحة';
                }

                var loginBox = document.querySelector('.login-box');
                if (loginBox) {
                    loginBox.style.animation = 'shake 0.5s';
                    setTimeout(function() {
                        loginBox.style.animation = '';
                    }, 500);
                }
            }
        }

        // Logout function
        function doLogout() {
            var loginScreen = document.getElementById('loginScreen');
            var mainApp = document.getElementById('mainApp');
            var passwordInput = document.getElementById('password');

            if (mainApp) mainApp.classList.add('hidden');
            if (loginScreen) loginScreen.classList.remove('hidden');
            if (passwordInput) passwordInput.value = '';

            stopCamera();
        }

        // Camera Functions
        function openCamera() {
            if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
                showModal('خطأ', 'متصفحك لا يدعم الكاميرا');
                return;
            }

            navigator.mediaDevices.getUserMedia({ 
                video: { 
                    facingMode: 'user',
                    width: { ideal: 1280 },
                    height: { ideal: 720 }
                } 
            }).then(function(mediaStream) {
                stream = mediaStream;
                var video = document.getElementById('video');
                if (video) {
                    video.srcObject = stream;
                }

                var overlay = document.getElementById('cameraOverlay');
                var captureBtn = document.getElementById('captureBtn');
                var openBtn = document.getElementById('openCamBtn');

                if (overlay) overlay.classList.add('hidden');
                if (captureBtn) captureBtn.classList.remove('hidden');
                if (openBtn) openBtn.classList.add('hidden');

                currentPhotoIndex = 0;
                updatePhotoCounter();
                updateCameraStatus();
            }).catch(function(err) {
                showModal('خطأ', 'لا يمكن الوصول إلى الكاميرا. تأكد من منح الإذن.');
            });
        }

        function updatePhotoCounter() {
            var counter = document.getElementById('photoCounter');
            if (counter) counter.textContent = currentPhotoIndex + 1;
        }

        function updateCameraStatus() {
            var status = document.getElementById('cameraStatus');
            if (!status) return;

            var capturedCount = 0;
            for (var i = 0; i < 3; i++) {
                if (capturedPhotos[i] !== null) capturedCount++;
            }

            if (capturedCount === 0) {
                status.textContent = 'لم يتم التقاط أي صورة بعد - اضغط التقاط للصورة الأولى';
                status.style.color = '#888';
            } else if (capturedCount < 3) {
                status.textContent = 'تم التقاط ' + capturedCount + ' من 3 صور - الصورة التالية: ' + (currentPhotoIndex + 1);
                status.style.color = '#e94560';
            } else {
                status.textContent = 'تم التقاط جميع الصور الثلاث بنجاح';
                status.style.color = '#2d6a4f';
            }
        }

        function capturePhoto() {
            if (currentPhotoIndex >= 3) return;

            var video = document.getElementById('video');
            var canvas = document.getElementById('canvas');

            if (!video || !canvas) return;

            canvas.width = video.videoWidth || 640;
            canvas.height = video.videoHeight || 480;
            var ctx = canvas.getContext('2d');

            ctx.save();
            ctx.translate(canvas.width/2, canvas.height/2);

            if (currentPhotoIndex === 0) {
                ctx.scale(1.05, 1.05);
                ctx.translate(-canvas.width/2, -canvas.height/2);
                ctx.drawImage(video, 0, 0);
            } else if (currentPhotoIndex === 1) {
                ctx.rotate(-0.1);
                ctx.scale(1.08, 1.02);
                ctx.translate(-canvas.width/2, -canvas.height/2);
                ctx.drawImage(video, 0, 0);
            } else {
                ctx.rotate(0.1);
                ctx.scale(1.08, 1.02);
                ctx.translate(-canvas.width/2, -canvas.height/2);
                ctx.drawImage(video, 0, 0);
            }

            ctx.restore();

            var photoData = canvas.toDataURL('image/jpeg', 0.9);
            capturedPhotos[currentPhotoIndex] = photoData;

            var imgElement = document.getElementById('capturedImage' + (currentPhotoIndex + 1));
            if (imgElement) {
                imgElement.src = photoData;
                imgElement.style.display = 'block';
            }

            var slot = document.getElementById('photoSlot' + (currentPhotoIndex + 1));
            if (slot) slot.classList.add('has-photo');

            currentPhotoIndex++;
            updatePhotoCounter();
            updateCameraStatus();

            if (currentPhotoIndex >= 3) {
                var captureBtn = document.getElementById('captureBtn');
                var retakeBtn = document.getElementById('retakeBtn');
                if (captureBtn) captureBtn.classList.add('hidden');
                if (retakeBtn) retakeBtn.classList.remove('hidden');
                stopCamera();
            }
        }

        function retakePhoto() {
            for (var i = 1; i <= 3; i++) {
                var img = document.getElementById('capturedImage' + i);
                var slot = document.getElementById('photoSlot' + i);
                if (img) {
                    img.style.display = 'none';
                    img.src = '';
                }
                if (slot) slot.classList.remove('has-photo');
            }
            capturedPhotos = [null, null, null];
            currentPhotoIndex = 0;

            var retakeBtn = document.getElementById('retakeBtn');
            var captureBtn = document.getElementById('captureBtn');
            var openBtn = document.getElementById('openCamBtn');

            if (retakeBtn) retakeBtn.classList.add('hidden');
            if (captureBtn) captureBtn.classList.add('hidden');
            if (openBtn) openBtn.classList.remove('hidden');

            updateCameraStatus();
        }

        function stopCamera() {
            if (stream) {
                var tracks = stream.getTracks();
                for (var i = 0; i < tracks.length; i++) {
                    tracks[i].stop();
                }
                stream = null;
            }
            var video = document.getElementById('video');
            if (video) video.srcObject = null;
        }

        // File Handling
        function handleFiles(files) {
            for (var i = 0; i < files.length; i++) {
                var file = files[i];
                var reader = new FileReader();
                reader.onload = (function(f) {
                    return function(e) {
                        currentFiles.push({
                            name: f.name,
                            type: f.type,
                            data: e.target.result,
                            size: f.size
                        });
                        updateFilesList();
                    };
                })(file);
                reader.readAsDataURL(file);
            }
        }

        function updateFilesList() {
            var container = document.getElementById('scannedFiles');
            if (!container) return;
            container.innerHTML = '';

            for (var i = 0; i < currentFiles.length; i++) {
                var div = document.createElement('div');
                div.className = 'file-item';
                div.innerHTML = '<span>📎 ' + currentFiles[i].name + '</span><span class="remove-file" onclick="removeFile(' + i + ')">✕</span>';
                container.appendChild(div);
            }
        }

        function removeFile(index) {
            currentFiles.splice(index, 1);
            updateFilesList();
        }

        // Save Record
        function saveRecord() {
            var record = {
                id: Date.now(),
                accusedName: document.getElementById('accusedName') ? document.getElementById('accusedName').value : '',
                legalArticle: document.getElementById('legalArticle') ? document.getElementById('legalArticle').value : '',
                departmentName: document.getElementById('departmentName') ? document.getElementById('departmentName').value : '',
                baseNumber: document.getElementById('baseNumber') ? document.getElementById('baseNumber').value : '',
                formNumber: document.getElementById('formNumber') ? document.getElementById('formNumber').value : '',
                photos: capturedPhotos,
                files: currentFiles,
                date: new Date().toLocaleString('ar-IQ')
            };

            if (!record.accusedName) {
                showModal('تنبيه', 'يرجى إدخال اسم المتهم على الأقل');
                return;
            }

            records.push(record);
            try {
                localStorage.setItem('criminalRecords', JSON.stringify(records));
            } catch(e) {
                showModal('خطأ', 'لا يمكن الحفظ - مساحة التخزين ممتلئة');
                return;
            }

            showModal('تم الحفظ', 'تم حفظ الاستمارة في المحفوظات بنجاح');
            clearForm();
            loadRecords();
        }

        function clearForm() {
            var accusedName = document.getElementById('accusedName');
            var legalArticle = document.getElementById('legalArticle');
            var departmentName = document.getElementById('departmentName');
            var baseNumber = document.getElementById('baseNumber');
            var formNumber = document.getElementById('formNumber');

            if (accusedName) accusedName.value = '';
            if (legalArticle) legalArticle.value = '';
            if (departmentName) departmentName.value = '';
            if (baseNumber) baseNumber.value = '';
            if (formNumber) formNumber.value = '';

            for (var i = 1; i <= 3; i++) {
                var img = document.getElementById('capturedImage' + i);
                var slot = document.getElementById('photoSlot' + i);
                if (img) {
                    img.style.display = 'none';
                    img.src = '';
                }
                if (slot) slot.classList.remove('has-photo');
            }
            capturedPhotos = [null, null, null];
            currentPhotoIndex = 0;

            var retakeBtn = document.getElementById('retakeBtn');
            var captureBtn = document.getElementById('captureBtn');
            var openBtn = document.getElementById('openCamBtn');

            if (retakeBtn) retakeBtn.classList.add('hidden');
            if (captureBtn) captureBtn.classList.add('hidden');
            if (openBtn) openBtn.classList.remove('hidden');

            updateCameraStatus();
            currentFiles = [];
            updateFilesList();
        }

        // Load Records
        function loadRecords() {
            var container = document.getElementById('recordsList');
            if (!container) return;
            container.innerHTML = '';

            if (records.length === 0) {
                container.innerHTML = '<p style="color:#888;text-align:center;">لا توجد استمارات محفوظة في المحفوظات</p>';
                return;
            }

            for (var i = records.length - 1; i >= 0; i--) {
                var record = records[i];
                var photoCount = 0;
                if (record.photos) {
                    for (var j = 0; j < 3; j++) {
                        if (record.photos[j]) photoCount++;
                    }
                }

                var div = document.createElement('div');
                div.className = 'record-item';
                div.onclick = (function(id) {
                    return function() { loadRecord(id); };
                })(record.id);

                div.innerHTML = '<div class="record-header"><h3>' + (record.accusedName || 'بدون اسم') + '</h3><span class="record-date">' + record.date + '</span></div>' +
                    '<div class="record-details"><span><strong>المادة:</strong> ' + (record.legalArticle || '-') + '</span>' +
                    '<span><strong>الشعبة:</strong> ' + (record.departmentName || '-') + '</span>' +
                    '<span><strong>رقم الأساس:</strong> ' + (record.baseNumber || '-') + '</span>' +
                    '<span><strong>رقم الاستمارة:</strong> ' + (record.formNumber || '-') + '</span>' +
                    '<span><strong>الصور:</strong> ' + photoCount + '/3</span>' +
                    '<span><strong>الملفات:</strong> ' + (record.files ? record.files.length : 0) + '</span></div>';

                container.appendChild(div);
            }
        }

        function loadRecord(id) {
            var record = null;
            for (var i = 0; i < records.length; i++) {
                if (records[i].id === id) {
                    record = records[i];
                    break;
                }
            }
            if (!record) return;

            var accusedName = document.getElementById('accusedName');
            var legalArticle = document.getElementById('legalArticle');
            var departmentName = document.getElementById('departmentName');
            var baseNumber = document.getElementById('baseNumber');
            var formNumber = document.getElementById('formNumber');

            if (accusedName) accusedName.value = record.accusedName || '';
            if (legalArticle) legalArticle.value = record.legalArticle || '';
            if (departmentName) departmentName.value = record.departmentName || '';
            if (baseNumber) baseNumber.value = record.baseNumber || '';
            if (formNumber) formNumber.value = record.formNumber || '';

            if (record.photos && record.photos.length > 0) {
                capturedPhotos = record.photos;
                for (var i = 0; i < 3; i++) {
                    if (record.photos[i]) {
                        var img = document.getElementById('capturedImage' + (i + 1));
                        var slot = document.getElementById('photoSlot' + (i + 1));
                        if (img) {
                            img.src = record.photos[i];
                            img.style.display = 'block';
                        }
                        if (slot) slot.classList.add('has-photo');
                    }
                }
                currentPhotoIndex = 3;

                var retakeBtn = document.getElementById('retakeBtn');
                var captureBtn = document.getElementById('captureBtn');
                var openBtn = document.getElementById('openCamBtn');

                if (retakeBtn) retakeBtn.classList.remove('hidden');
                if (captureBtn) captureBtn.classList.add('hidden');
                if (openBtn) openBtn.classList.add('hidden');

                updateCameraStatus();
            }

            currentFiles = record.files || [];
            updateFilesList();

            window.scrollTo(0, 0);
        }

        // Print
        function printForm() {
            var printDate = document.getElementById('printDate');
            if (printDate) {
                printDate.textContent = 'تاريخ الإصدار: ' + new Date().toLocaleString('ar-IQ');
            }
            window.print();
        }

        // Backup
        function backupData() {
            var data = {
                records: records,
                backupDate: new Date().toISOString()
            };
            var blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' };
            var url = URL.createObjectURL(blob);
            var a = document.createElement('a');
            a.href = url;
            a.download = 'criminal_backup_' + Date.now() + '.json';
            a.click();
            URL.revokeObjectURL(url);
            showModal('تم', 'تم إنشاء النسخة الاحتياطية بنجاح');
        }

        // Restore
        function restoreData() {
            var input = document.createElement('input');
            input.type = 'file';
            input.accept = '.json';
            input.onchange = function(e) {
                var file = e.target.files[0];
                if (!file) return;

                var reader = new FileReader();
                reader.onload = function(event) {
                    try {
                        var data = JSON.parse(event.target.result);
                        if (data.records) {
                            records = data.records;
                            try {
                                localStorage.setItem('criminalRecords', JSON.stringify(records));
                            } catch(e) {
                                showModal('خطأ', 'لا يمكن الحفظ في التخزين المحلي');
                                return;
                            }
                            loadRecords();
                            showModal('تم', 'تم استعادة البيانات بنجاح');
                        }
                    } catch (err) {
                        showModal('خطأ', 'ملف النسخة الاحتياطية غير صالح');
                    }
                };
                reader.readAsText(file);
            };
            input.click();
        }

        // Modal
        function showModal(title, message) {
            var modalTitle = document.getElementById('modalTitle');
            var modalMessage = document.getElementById('modalMessage');
            var modalOverlay = document.getElementById('modalOverlay');

            if (modalTitle) modalTitle.textContent = title;
            if (modalMessage) modalMessage.textContent = message;
            if (modalOverlay) modalOverlay.classList.add('active');
        }

        function closeModal() {
            var modalOverlay = document.getElementById('modalOverlay');
            if (modalOverlay) modalOverlay.classList.remove('active');
        }
